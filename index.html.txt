<script>

const contractAddress = "0x80396aD5F3d2a234691e191944d0D903F16bD4C8";

const abi = [
    "function setMessage(string)",
    "function getMessage() view returns (string)",
    "function getOwner() view returns (address)",
    "function clearMessage()"
];

let contract;

async function connectWallet() {
    try {
        if (!window.ethereum) {
            alert("Please install MetaMask");
            return;
        }

        const provider = new ethers.BrowserProvider(window.ethereum);
        await provider.send("eth_requestAccounts", []);

        const network = await provider.getNetwork();

        if (network.chainId !== 11155111n) {
            alert("Please switch to Sepolia");
            return;
        }

        const signer = await provider.getSigner();

        contract = new ethers.Contract(
            contractAddress, abi, signer
        );

        document.getElementById("wallet").innerText =
            "Connected: " + await signer.getAddress();

    } catch (error) {
        console.error(error);
        alert("Wallet connection failed");
    }
}

async function readMessage() {
    try {
        if (!contract) {
            alert("Connect your wallet first");
            return;
        }

        const message = await contract.getMessage();

        document.getElementById("currentMessage").innerText = message;

    } catch (error) {
        console.error(error);
        alert("Could not read message");
    }
}

async function saveMessage() {
    try {
        if (!contract) {
            alert("Connect your wallet first");
            return;
        }

        const message = document.getElementById("messageInput").value;

        if (!message.trim()) {
            alert("Enter a message first");
            return;
        }

        document.getElementById("result").innerText =
            "Waiting for confirmation...";

        const tx = await contract.setMessage(message);

        document.getElementById("result").innerText =
            "Transaction: " + tx.hash;

        await tx.wait();

        document.getElementById("result").innerText =
            "Transaction confirmed!";

        await readMessage();

    } catch (error) {
        console.error(error);
        document.getElementById("result").innerText =
            "Transaction failed or rejected";
    }
}

</script>
